PostgreSQL Database Schema

-- Create database
CREATE DATABASE estate_link;

-- Switch to database
\c estate_link;

-- Users table stores all accounts: customers, hosts, admins
CREATE TABLE users (
  id SERIAL PRIMARY KEY,
  external_id VARCHAR(50) UNIQUE,
  full_name VARCHAR(255) NOT NULL,
  email VARCHAR(255) UNIQUE NOT NULL,
  phone VARCHAR(50),
  role VARCHAR(40) NOT NULL DEFAULT 'user',
  status VARCHAR(40) NOT NULL DEFAULT 'active',
  avatar_url TEXT,
  password_hash TEXT,
  bookings_count INT DEFAULT 0,
  joined_at TIMESTAMP WITH TIME ZONE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Host profile data extends a user account when the user is a host
CREATE TABLE host_profiles (
  id SERIAL PRIMARY KEY,
  user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  host_code VARCHAR(50) UNIQUE,
  joined_at TIMESTAMP WITH TIME ZONE,
  status VARCHAR(40) NOT NULL DEFAULT 'pending',
  listings_count INT DEFAULT 0,
  earnings NUMERIC(14,2) DEFAULT 0,
  rating NUMERIC(3,2),
  verified BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Properties available for rent, lease, or purchase
CREATE TABLE properties (
  id SERIAL PRIMARY KEY,
  host_id INT NOT NULL REFERENCES host_profiles(id) ON DELETE CASCADE,
  title VARCHAR(255) NOT NULL,
  location VARCHAR(255),
  category VARCHAR(100),
  listing_type VARCHAR(50),
  beds SMALLINT,
  bedrooms SMALLINT,
  baths SMALLINT,
  guests SMALLINT,
  price NUMERIC(14,2),
  period VARCHAR(50),
  rating NUMERIC(3,2),
  verified BOOLEAN DEFAULT FALSE,
  main_image TEXT,
  description TEXT,
  is_available BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE property_images (
  id SERIAL PRIMARY KEY,
  property_id INT NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
  image_url TEXT NOT NULL,
  sort_order SMALLINT DEFAULT 0
);

CREATE TABLE property_amenities (
  id SERIAL PRIMARY KEY,
  property_id INT NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
  amenity VARCHAR(150) NOT NULL
);

-- Booking records for rent and lease reservations
CREATE TABLE bookings (
  id SERIAL PRIMARY KEY,
  property_id INT NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
  user_id INT REFERENCES users(id) ON DELETE SET NULL,
  check_in DATE NOT NULL,
  check_out DATE NOT NULL,
  guests SMALLINT NOT NULL,
  guest_name VARCHAR(255),
  guest_email VARCHAR(255),
  status VARCHAR(50) NOT NULL DEFAULT 'pending',
  total_amount NUMERIC(14,2),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Purchase requests for properties listed for sale
CREATE TABLE purchase_requests (
  id SERIAL PRIMARY KEY,
  property_id INT NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
  user_id INT REFERENCES users(id) ON DELETE SET NULL,
  requested_date DATE,
  offer_price NUMERIC(14,2) NOT NULL,
  message TEXT,
  status VARCHAR(50) NOT NULL DEFAULT 'pending',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Host verification requests and document uploads
CREATE TABLE host_verifications (
  id SERIAL PRIMARY KEY,
  host_id INT NOT NULL REFERENCES host_profiles(id) ON DELETE CASCADE,
  document_type VARCHAR(100),
  document_number VARCHAR(100),
  document_front_url TEXT,
  document_back_url TEXT,
  selfie_image_url TEXT,
  submitted_at TIMESTAMP WITH TIME ZONE,
  status VARCHAR(50) NOT NULL DEFAULT 'pending',
  property_count INT DEFAULT 0,
  rejection_reason TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Conversation threads between host and guest
CREATE TABLE conversations (
  id SERIAL PRIMARY KEY,
  property_id INT REFERENCES properties(id) ON DELETE SET NULL,
  host_id INT REFERENCES host_profiles(id) ON DELETE SET NULL,
  guest_id INT REFERENCES users(id) ON DELETE SET NULL,
  last_message TEXT,
  last_message_at TIMESTAMP WITH TIME ZONE,
  unread_count INT DEFAULT 0,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE messages (
  id SERIAL PRIMARY KEY,
  conversation_id INT NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
  sender VARCHAR(50) NOT NULL,
  text TEXT NOT NULL,
  sent_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Wishlist entries for user saved properties
CREATE TABLE wishlists (
  id SERIAL PRIMARY KEY,
  user_id INT NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  property_id INT NOT NULL REFERENCES properties(id) ON DELETE CASCADE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  UNIQUE(user_id, property_id)
);

-- Reports for users, hosts, and properties
CREATE TABLE reports (
  id SERIAL PRIMARY KEY,
  reporter_id INT REFERENCES users(id) ON DELETE SET NULL,
  target_type VARCHAR(50) NOT NULL,
  target_id INT NOT NULL,
  issue_type VARCHAR(150),
  description TEXT,
  status VARCHAR(50) NOT NULL DEFAULT 'open',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Payout account details for hosts
CREATE TABLE payout_accounts (
  id SERIAL PRIMARY KEY,
  host_id INT NOT NULL REFERENCES host_profiles(id) ON DELETE CASCADE,
  account_holder VARCHAR(255),
  bank_name VARCHAR(255),
  account_number VARCHAR(100),
  iban VARCHAR(50),
  swift VARCHAR(50),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Payment transactions tied to bookings, purchase requests, and properties
CREATE TABLE payments (
  id SERIAL PRIMARY KEY,
  booking_id INT REFERENCES bookings(id) ON DELETE SET NULL,
  purchase_request_id INT REFERENCES purchase_requests(id) ON DELETE SET NULL,
  user_id INT REFERENCES users(id) ON DELETE SET NULL,
  property_id INT REFERENCES properties(id) ON DELETE SET NULL,
  amount NUMERIC(14,2) NOT NULL,
  payment_type VARCHAR(50),
  payment_method VARCHAR(50),
  status VARCHAR(50) NOT NULL DEFAULT 'pending',
  paid_at TIMESTAMP WITH TIME ZONE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

```sql
-- ============================================================
-- ESTATE LINK DATABASE

-- ============================================================
-- 1. USERS
-- ============================================================

CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,

    external_id VARCHAR(50) UNIQUE,

    full_name VARCHAR(255) NOT NULL,

    email VARCHAR(255) NOT NULL UNIQUE,

    phone VARCHAR(50),

    role VARCHAR(30) NOT NULL DEFAULT 'user'
        CHECK (role IN ('user', 'host', 'admin')),

    status VARCHAR(30) NOT NULL DEFAULT 'active'
        CHECK (status IN (
            'active',
            'inactive',
            'suspended',
            'pending'
        )),

    avatar_url TEXT,

    password_hash TEXT,

    joined_at TIMESTAMPTZ DEFAULT NOW(),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 2. HOST PROFILES
-- One user can have one host profile.
-- ============================================================

CREATE TABLE host_profiles (
    id BIGSERIAL PRIMARY KEY,

    user_id BIGINT NOT NULL UNIQUE
        REFERENCES users(id)
        ON DELETE CASCADE,

    host_code VARCHAR(50) UNIQUE,

    status VARCHAR(30) NOT NULL DEFAULT 'pending'
        CHECK (status IN (
            'pending',
            'active',
            'suspended',
            'rejected'
        )),

    verified BOOLEAN NOT NULL DEFAULT FALSE,

    joined_at TIMESTAMPTZ DEFAULT NOW(),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 3. PROPERTIES
-- ============================================================

CREATE TABLE properties (
    id BIGSERIAL PRIMARY KEY,

    host_id BIGINT NOT NULL
        REFERENCES host_profiles(id)
        ON DELETE CASCADE,

    title VARCHAR(255) NOT NULL,

    description TEXT,

    property_type VARCHAR(50) NOT NULL
        CHECK (property_type IN (
            'hotel',
            'apartment',
            'house',
            'villa',
            'plot',
            'office',
            'commercial',
            'other'
        )),

    listing_type VARCHAR(30) NOT NULL
        CHECK (listing_type IN (
            'rent',
            'sale',
            'lease'
        )),

    address TEXT,

    city VARCHAR(100),

    state VARCHAR(100),

    country VARCHAR(100),

    latitude NUMERIC(10, 7),

    longitude NUMERIC(10, 7),

    beds SMALLINT CHECK (beds IS NULL OR beds >= 0),

    bedrooms SMALLINT CHECK (bedrooms IS NULL OR bedrooms >= 0),

    baths SMALLINT CHECK (baths IS NULL OR baths >= 0),

    guests SMALLINT CHECK (guests IS NULL OR guests >= 0),

    price NUMERIC(14, 2) NOT NULL
        CHECK (price >= 0),

    currency VARCHAR(10) NOT NULL DEFAULT 'USD',

    price_period VARCHAR(30)
        CHECK (price_period IN (
            'night',
            'day',
            'week',
            'month',
            'year',
            'total'
        )),

    status VARCHAR(30) NOT NULL DEFAULT 'pending'
        CHECK (status IN (
            'draft',
            'pending',
            'active',
            'rejected',
            'suspended',
            'sold',
            'inactive'
        )),

    verified BOOLEAN NOT NULL DEFAULT FALSE,

    main_image_url TEXT,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 4. PROPERTY IMAGES
-- ============================================================

CREATE TABLE property_images (
    id BIGSERIAL PRIMARY KEY,

    property_id BIGINT NOT NULL
        REFERENCES properties(id)
        ON DELETE CASCADE,

    image_url TEXT NOT NULL,

    sort_order INT NOT NULL DEFAULT 0,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 5. PROPERTY AMENITIES
-- ============================================================

CREATE TABLE property_amenities (
    id BIGSERIAL PRIMARY KEY,

    property_id BIGINT NOT NULL
        REFERENCES properties(id)
        ON DELETE CASCADE,

    amenity VARCHAR(150) NOT NULL,

    UNIQUE(property_id, amenity)
);


-- ============================================================
-- 6. BOOKINGS
-- Used for rental / lease reservations.
-- ============================================================

CREATE TABLE bookings (
    id BIGSERIAL PRIMARY KEY,

    property_id BIGINT NOT NULL
        REFERENCES properties(id)
        ON DELETE RESTRICT,

    user_id BIGINT
        REFERENCES users(id)
        ON DELETE SET NULL,

    check_in DATE NOT NULL,

    check_out DATE NOT NULL,

    guests SMALLINT NOT NULL
        CHECK (guests > 0),

    guest_name VARCHAR(255),

    guest_email VARCHAR(255),

    status VARCHAR(30) NOT NULL DEFAULT 'pending'
        CHECK (status IN (
            'pending',
            'confirmed',
            'cancelled',
            'rejected',
            'completed'
        )),

    total_amount NUMERIC(14, 2)
        CHECK (total_amount IS NULL OR total_amount >= 0),

    currency VARCHAR(10) NOT NULL DEFAULT 'USD',

    cancelled_at TIMESTAMPTZ,

    confirmed_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    CHECK (check_out > check_in)
);


-- ============================================================
-- 7. PURCHASE REQUESTS
-- Used when a user wants to purchase a property.
-- ============================================================

CREATE TABLE purchase_requests (
    id BIGSERIAL PRIMARY KEY,

    property_id BIGINT NOT NULL
        REFERENCES properties(id)
        ON DELETE RESTRICT,

    user_id BIGINT
        REFERENCES users(id)
        ON DELETE SET NULL,

    requested_date DATE NOT NULL DEFAULT CURRENT_DATE,

    offer_price NUMERIC(14, 2) NOT NULL
        CHECK (offer_price > 0),

    currency VARCHAR(10) NOT NULL DEFAULT 'USD',

    message TEXT,

    status VARCHAR(30) NOT NULL DEFAULT 'pending'
        CHECK (status IN (
            'pending',
            'accepted',
            'rejected',
            'withdrawn',
            'expired',
            'completed'
        )),

    responded_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 8. HOST VERIFICATIONS
-- ============================================================

CREATE TABLE host_verifications (
    id BIGSERIAL PRIMARY KEY,

    host_id BIGINT NOT NULL
        REFERENCES host_profiles(id)
        ON DELETE CASCADE,

    document_type VARCHAR(100) NOT NULL,

    document_number VARCHAR(100),

    document_front_url TEXT,

    document_back_url TEXT,

    selfie_image_url TEXT,

    status VARCHAR(30) NOT NULL DEFAULT 'pending'
        CHECK (status IN (
            'pending',
            'approved',
            'rejected'
        )),

    rejection_reason TEXT,

    submitted_at TIMESTAMPTZ DEFAULT NOW(),

    reviewed_by BIGINT
        REFERENCES users(id)
        ON DELETE SET NULL,

    reviewed_at TIMESTAMPTZ,

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 9. CONVERSATIONS
-- Host <-> Guest conversation for a property.
-- ============================================================

CREATE TABLE conversations (
    id BIGSERIAL PRIMARY KEY,

    property_id BIGINT
        REFERENCES properties(id)
        ON DELETE SET NULL,

    host_id BIGINT
        REFERENCES host_profiles(id)
        ON DELETE SET NULL,

    guest_id BIGINT
        REFERENCES users(id)
        ON DELETE SET NULL,

    last_message TEXT,

    last_message_at TIMESTAMPTZ,

    unread_count INT NOT NULL DEFAULT 0
        CHECK (unread_count >= 0),

    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);


-- ============================================================
-- 10. MESSAGES
-- ============================================================

CREATE TABLE messages (
    id BIGSERIAL PRIMARY KEY,

    conversation_id BIGINT NOT NULL
        REFERENCES conversations(id)
        ON DELETE CASCADE,

    sender_id BIGINT NOT NULL
        REFERENCES users(id)
        ON DELETE CASCADE,

    message_text TEXT NOT NULL,

    sent_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

    read_at TIMESTAMPTZ

CREATE TABLE wishlists (

-- ============================================================
-- 12. REVIEWS

CREATE TABLE reviews (

    user_id BIGINT
        REFERENCES users(id)
        ON DELETE SET NULL,

    booking_id BIGINT
        REFERENCES bookings(id)

    rating SMALLINT NOT NULL

    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

-- UPDATED_AT TRIGGER
-- Automatically updates updated_at fields.
-- ============================================================

CREATE OR REPLACE FUNCTION update_updated_at_column()
RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;


CREATE TRIGGER update_users_updated_at
BEFORE UPDATE ON users
FOR EACH ROW
```
